57    fn transform_module_info(&self, module_info: &mut ModuleInfo) -> Result<(), MiddlewareError> {
58        let current = Instant::now();
59
60        let mut global_indexes = self.global_indexes.lock().unwrap();
61        if global_indexes.is_some() {
62            panic!("GasCalibration::transform_module_info: Attempting to use a `GasCalibration` middleware from multiple modules.");
63        }
64
65        // println!("{:?}", global_indexes.transform_module_info_ms);
66
67        let mut indexes = GasCalibrationGlobalIndexes {
68            imports_call_map: Default::default(),
69            op_call_map: Default::default(),
70            // param_size_current: global_index,
71            param_size_map: Default::default(),
72        };
73
74        for (import_key, import_index) in module_info.imports.iter() {
75            let module_name = import_key.module.clone();
76            let function_name = import_key.field.clone();
77            let index = import_key.import_idx;
78
79            // -> env.abort OR massa.assembly_script_print
80            let function_fullname = format!("{}.{}", module_name, function_name);
81
82            // Append a global for this 'imports' (== abi call) and initialize
83            // it.
84            let global_index = module_info
85                .globals
86                .push(GlobalType::new(Type::I64, Mutability::Var));
87            module_info
88                .global_initializers
89                .push(GlobalInit::I64Const(0));
90            module_info.exports.insert(
91                format!("wgc_abi_{}", function_fullname),
92                ExportIndex::Global(global_index),
93            );
94
95            indexes
96                .imports_call_map
97                .insert(index, (function_fullname.clone(), global_index));
98
99            // Append a global per param size per 'imports' (== abi call)
100            let function_sig = if let ImportIndex::Function(f) = import_index {
101                module_info.functions.get(*f).unwrap_or_else(|| {
102                    panic!("Cannot get function signature for {}", function_name)
103                })
104            } else {
105                panic!("ImportIndex is not a function??");
106            };
107            let function_type = module_info.signatures.get(*function_sig).unwrap();
108            let param_count = function_type.params().len();
109
110            for i in 0..param_count {
111                let global_index = module_info
112                    .globals
113                    .push(GlobalType::new(Type::I64, Mutability::Var));
114                module_info
115                    .global_initializers
116                    .push(GlobalInit::I64Const(0));
117
118                module_info.exports.insert(
119                    format!("wgc_ps_{}:{}", function_fullname, i),
120                    ExportIndex::Global(global_index),
121                );
122
123                indexes.param_size_map.insert(index, global_index);
124            }
125        }
126
127        for op_name in OPERATOR_VARIANTS {
128            // Append a global for this operator and initialize it.
129            let global_index = module_info
130                .globals
131                .push(GlobalType::new(Type::I64, Mutability::Var));
132            module_info
133                .global_initializers
134                .push(GlobalInit::I64Const(0));
135            module_info.exports.insert(
136                format!("wgc_op_{}", op_name),
137                ExportIndex::Global(global_index),
138            );
139
140            indexes
141                .op_call_map
142                .insert((*op_name).to_string(), global_index);
143        }
144
145        // println!("module info function names: {:?}",
146        // module_info.function_names); println!("module info exports:
147        // {:?}", module_info.exports); println!("module info imports:
148        // {:?}", module_info.imports); println!("module info functions:
149        // {:?}", module_info.functions); println!("module info num
150        // imported functions: {:?}", module_info.num_imported_functions);
151        // println!("module info start function: {:?}",
152        // module_info.start_function); println!("module info passive
153        // elements: {:?}", module_info.passive_elements);
154        // println!("module info signatures: {:?}", module_info.signatures);
155
156        // Append a global variable for time elapsed of this function.
157        // Note: transform_module_info can take quite some time and thus we need
158        // this timing for accurate gas calibration
159        let global_index = module_info
160            .globals
161            .push(GlobalType::new(Type::F64, Mutability::Var));
162        module_info
163            .global_initializers
164            .push(GlobalInit::F64Const(current.elapsed().as_secs_f64()));
165        module_info.exports.insert(
166            String::from("wgc_elapsed_transform_module_info"),
167            ExportIndex::Global(global_index),
168        );
169
170        // indexes.transform_module_info_ms += duration.as_millis() as f64;
171        // println!("Time elapsed in {}() is: {:?}", "transform_module_info",
172        // duration);
173
174        *global_indexes = Some(indexes);
175
176        Ok(())
177    }
178}

180impl FunctionMiddleware for FunctionGasCalibration {
181    fn feed<'a>(
182        &mut self,
183        operator: Operator<'a>,
184        state: &mut MiddlewareReaderState<'a>,
185    ) -> Result<(), MiddlewareError> {
186        // let current = Instant::now();
187
188        // println!("Operator: {:?}", operator);
189        state.push_operator(operator.clone());
190
191        if let Operator::Call { function_index } = operator {
192            // let f = self.global_indexes.imports_call_map.get(&
193            // function_index).unwrap(); println!("Operator::Call
194            // {:?}", f);
195
196            // state.push_operator(operator);
197
198            if let Some(index) = self.global_indexes.imports_call_map.get(&function_index) {
199                // println!("Found function index: {}", function_index);
200                state.extend(&[
201                    // incr function call counter
202                    Operator::GlobalGet {
203                        global_index: index.1.as_u32(),
204                    },
205                    Operator::I64Const { value: 1_i64 },
206                    Operator::I64Add,
207                    Operator::GlobalSet {
208                        global_index: index.1.as_u32(),
209                    },
210                ]);
211            } else {
212                // Note: here we are skipping call to 'local function'
213                // For instance, getOpKeys() use derOpKeys() (local) +
214                // get_op_keys() (abi) Uncomment the line
215                // 'println!("...", module_info.functions);' to view the list of
216                // all functions (import + local)
217                // Note2: Signature of function (e.g. arguments types + return
218                // type) can be seen with: println!("...",
219                // module_info.signatures);
220
221                // println!("Skipping unknown function index: {}",
222                // function_index);
223            }
224        }
225
226        let op_name = operator_field_str(&operator);
227        let index = self
228            .global_indexes
229            .op_call_map
230            .get(op_name)
231            .ok_or_else(|| {
232                MiddlewareError::new(
233                    "GasCalibration",
234                    format!("Unable to get index for op: {}", op_name),
235                )
236            })?;
237
238        state.extend(&[
239            Operator::GlobalGet {
240                global_index: index.as_u32(),
241            },
242            Operator::I64Const { value: 1_i64 },
243            Operator::I64Add,
244            Operator::GlobalSet {
245                global_index: index.as_u32(),
246            },
247        ]);
248
249        // let duration = current.elapsed();
250        // println!("Time elapsed in {}() is: {:?}", "feed", duration);
251
252        Ok(())
253    }
254}

263pub fn get_gas_calibration_result(
264    instance: &Instance,
265    store: &mut impl AsStoreMut,
266) -> GasCalibrationResult {
267    let current = Instant::now();
268
269    let mut result = GasCalibrationResult {
270        counters: Default::default(),
271        timers: Default::default(),
272    };
273    let patterns = [
274        r"wgc_abi_([\w\.]+)",
275        r"wgc_op_([\w]+)",
276        r"wgc_ps_([\w\.:]+)",
277        r"wgc_elapsed_([\w\._]+)",
278    ];
279    // Must not fail
280    let set = RegexSet::new(patterns).unwrap();
281    // Compile each pattern independently.
282    let regexes: Vec<_> = set
283        .patterns()
284        .iter()
285        .map(|pat| Regex::new(pat).unwrap()) // never fail, already compiled in RegexSet
286        .collect();
287
288    // let mut abi_call_index = 0;
289    let mut exports_iter = instance.exports.iter();
290
291    loop {
292        let export_ = exports_iter.next();
293        if export_.is_none() {
294            break;
295        }
296        // Safe to unwrap (tested against is_none())
297        let (export_name, extern_) = export_.unwrap();
298
299        let counter_value: Option<i64> = match extern_ {
300            Extern::Global(g) => g.get(store).try_into().ok(),
301            _ => None,
302        };
303        let timer_value: Option<f64> = match extern_ {
304            Extern::Global(g) => g.get(store).try_into().ok(),
305            _ => None,
306        };
307
308        if counter_value.is_none() && timer_value.is_none() {
309            continue;
310        }
311
312        let matches = set.matches(export_name);
313
314        match export_name {
315            ex_name if matches.matched(0) => {
316                let rgx = &regexes[0];
317                let mut rgx_iter = rgx.captures_iter(ex_name);
318
319                if let Some(cap) = rgx_iter.next() {
320                    if let Some(abi_func_name) = cap.get(1) {
321                        result.counters.insert(
322                            format!("Abi:call:{}", abi_func_name.as_str()),
323                            counter_value.unwrap() as u64,
324                        );
325                    }
326                }
327            }
328            ex_name if matches.matched(1) => {
329                let rgx = &regexes[1];
330                let mut rgx_iter = rgx.captures_iter(ex_name);
331
332                if let Some(cap) = rgx_iter.next() {
333                    if let Some(op_name) = cap.get(1) {
334                        result.counters.insert(
335                            format!("Wasm:{}", op_name.as_str()),
336                            counter_value.unwrap() as u64,
337                        );
338                    }
339                }
340            }
341            ex_name if matches.matched(2) => {
342                let rgx = &regexes[2];
343                let mut rgx_iter = rgx.captures_iter(ex_name);
344
345                if let Some(cap) = rgx_iter.next() {
346                    if let Some(op_name) = cap.get(1) {
347                        result.counters.insert(
348                            format!("Abi:ps:{}", op_name.as_str()),
349                            counter_value.unwrap() as u64,
350                        );
351                    }
352                }
353            }
354            ex_name if matches.matched(3) => {
355                let rgx = &regexes[3];
356                let mut rgx_iter = rgx.captures_iter(ex_name);
357
358                if let Some(cap) = rgx_iter.next() {
359                    if let Some(fn_name) = cap.get(1) {
360                        result
361                            .timers
362                            .insert(format!("Time:{}", fn_name.as_str()), timer_value.unwrap());
363                    }
364                }
365            }
366            _ => {
367                println!("Unhandled export name: {}", export_name);
368            }
369        }
370    }
371
372    let duration = current.elapsed();
373    result.timers.insert(
374        String::from("Time:gas_calibration_result"),
375        duration.as_secs_f64(),
376    );
377
378    // println!("Time elapsed in {}() is: {:?}", "gas_calibration_result",
379    // duration);
380
381    result
382}
