1//! Provides the `__rust_probestack` symbol expected by `wasmer-vm` 4.x.
2//!
3//! Rust >= 1.88 stopped exporting `__rust_probestack` with C linkage (it became
4//! an internal, mangled item of `compiler_builtins`). `wasmer-vm` < 6.1.0 still
5//! declares it as `extern "C"` to fill its builtin function table, so linking
6//! fails on x86_64 targets that use it.
7//!
8//! `wasmer-vm` only reaches for this symbol on non-Windows x86/x86_64: on
9//! aarch64 it uses an empty probestack, and on Windows it uses `__chkstk` /
10//! `___chkstk_ms`. So this shim is gated to the targets that need it.
11//!
12//! TODO: remove this shim once wasmer is bumped to >= 6.1.0, which ships the
13//! symbol itself.
14
15// ELF flavour (Linux, BSD, ...).
16#[cfg(all(target_arch = "x86_64", not(windows), not(target_vendor = "apple")))]
17core::arch::global_asm!(
18    r#"
19    .pushsection .text.__rust_probestack
20    .globl __rust_probestack
21    .type  __rust_probestack, @function
22    .hidden __rust_probestack
23__rust_probestack:
24    .cfi_startproc
25    pushq  %rbp
26    .cfi_adjust_cfa_offset 8
27    .cfi_offset %rbp, -16
28    movq   %rsp, %rbp
29    .cfi_def_cfa_register %rbp
30
31    mov    %rax,%r11
32    cmp    $0x1000,%r11
33    jna    3f
342:
35    sub    $0x1000,%rsp
36    test   %rsp,8(%rsp)
37    sub    $0x1000,%r11
38    cmp    $0x1000,%r11
39    ja     2b
403:
41    sub    %r11,%rsp
42    test   %rsp,8(%rsp)
43    add    %rax,%rsp
44
45    leave
46    .cfi_def_cfa_register %rsp
47    .cfi_adjust_cfa_offset -8
48    ret
49    .cfi_endproc
50    .size __rust_probestack, . - __rust_probestack
51    .popsection
52    "#,
53    options(att_syntax)
54);

56// Mach-O flavour (x86_64 macOS). Same body, but Mach-O prefixes C symbols with
57// an underscore and does not understand the ELF-only `.type` / `.size` /
58// `.hidden` directives.
59#[cfg(all(target_arch = "x86_64", target_vendor = "apple"))]
60core::arch::global_asm!(
61    r#"
62    .section __TEXT,__text
63    .globl ___rust_probestack
64    .p2align 4
65___rust_probestack:

72
