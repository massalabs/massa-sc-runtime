37    /// Create a VM instance from the current module
38    pub(crate) fn create_vm_instance_and_init_env(
39        &mut self,
40        store: &mut Store,
41    ) -> VMResult<(Instance, FunctionEnv<ASEnv>, u64)> {
42        let (imports, mut fenv) = self.resolver(store);
43        match Instance::new(store, &self.module, &imports) {
44            Ok(instance) => {
45                self.init_with_instance(store, &instance, &mut fenv)?;
46                let post_init_points = if cfg!(not(feature = "gas_calibration")) {
47                    if let MeteringPoints::Remaining(points) =
48                        metering::get_remaining_points(store, &instance)
49                    {
50                        points
51                    } else {
52                        0
53                    }
54                } else {
55                    0
56                };
57                self.env
58                    .abi_enabled
59                    .store(true, std::sync::atomic::Ordering::Relaxed);
60                Ok((instance, fenv, post_init_points))
61            }
62            Err(err) => {
63                // Filter the error created by the metering middleware when
64                // there is not enough gas at initialization
65                if let InstantiationError::Start(ref e) = err {
66                    if let Some(trap) = e.clone().to_trap() {
67                        if trap == TrapCode::UnreachableCodeReached && e.trace().is_empty() {
68                            vm_bail!("Not enough gas, limit reached at initialization");
69                        }
70                    }
71                }
72                Err(err.into())
73            }
74        }
75    }

77    pub(crate) fn execution(
78        &self,
79        store: &mut Store,
80        instance: &Instance,
81        function: &str,
82        param: &[u8],
83    ) -> VMResult<Response> {
84        if cfg!(not(feature = "gas_calibration")) {
85            // Sub initial metering cost
86            let metering_initial_cost = self.env.get_gas_costs().launch_cost;
87            let remaining_gas = get_remaining_points(&self.env, store)?;
88            if metering_initial_cost > remaining_gas {
89                vm_bail!("Not enough gas to launch the virtual machine")
90            }
91            set_remaining_points(&self.env, store, remaining_gas - metering_initial_cost)?;
92        }
93        // Now can exec
94        let wasm_func = instance.exports.get_function(function)?;
95        let argc = wasm_func.param_arity(store);
96        let res = if argc == 0 {
97            wasm_func.call(store, &[])
98        } else if argc == 1 {
99            let param_ptr = *BufferPtr::alloc(&param.to_vec(), self.env.get_ffi_env(), store)?;
100            wasm_func.call(store, &[Value::I32(param_ptr.offset() as i32)])
101        } else {
102            vm_bail!("Unexpected number of parameters in the function called")
103        };
104
105        match res {
106            Ok(value) => {
107                if function.eq(crate::settings::MAIN) {
108                    let remaining_gas = if cfg!(feature = "gas_calibration") {
109                        Ok(0_u64)
110                    } else {
111                        get_remaining_points(&self.env, store)
112                    };
113
114                    return Ok(Response {
115                        ret: Vec::new(), // main return empty vec
116                        remaining_gas: remaining_gas?,
117                        init_gas_cost: 0,
118                        #[cfg(feature = "execution-trace")]
119                        trace: Default::default(),
120                    });
121                }
122                let ret = if let Some(offset) = value.first() {
123                    if let Some(offset) = offset.i32() {
124                        let buffer_ptr = BufferPtr::new(offset as u32);
125                        let memory = instance.exports.get_memory("memory")?;
126                        buffer_ptr.read(memory, store)?
127                    } else {
128                        vm_bail!("Execution wasn't in capacity to read the return value")
129                    }
130                } else {
131                    Vec::new()
132                };
133                let remaining_gas = if cfg!(feature = "gas_calibration") {
134                    Ok(0_u64)
135                } else {
136                    get_remaining_points(&self.env, store)
137                };
138                Ok(Response {
139                    ret,
140                    remaining_gas: remaining_gas?,
141                    init_gas_cost: 0,
142                    #[cfg(feature = "execution-trace")]
143                    trace: Default::default(),
144                })
145            }
146            Err(error) => Err(error.into()),
147        }
148    }

150    fn init_with_instance(
151        &mut self,
152        store: &mut Store,
153        instance: &Instance,
154        fenv: &mut FunctionEnv<ASEnv>,
155    ) -> VMResult<()> {
156        let memory = instance.exports.get_memory("memory")?;
157
158        // NOTE: only add functions (__new, ...) if these exists in wasm/wat
159        // files       so we can still exec some very basic wat files
160        let fn_new = instance
161            .exports
162            .get_typed_function::<(i32, i32), i32>(&store, "__new")
163            .ok();
164        let fn_pin = instance
165            .exports
166            .get_typed_function::<i32, i32>(&store, "__pin")
167            .ok();
168        let fn_unpin = instance
169            .exports
170            .get_typed_function::<i32, ()>(&store, "__unpin")
171            .ok();
172        let fn_collect = instance
173            .exports
174            .get_typed_function::<(), ()>(&store, "__collect")
175            .ok();
176
177        fenv.as_mut(store).get_ffi_env_as_mut().init_with(
178            Some(memory.clone()),
179            fn_new.clone(),
180            fn_pin.clone(),
181            fn_unpin.clone(),
182            fn_collect.clone(),
183        );
184
185        // Update self.env as well
186        self.env.get_ffi_env_as_mut().init_with(
187            Some(memory.clone()),
188            fn_new,
189            fn_pin,
190            fn_unpin,
191            fn_collect,
192        );
193
194        // Metering counters
195        if cfg!(not(feature = "gas_calibration")) {
196            let g_1 = instance
197                .exports
198                .get_global("wasmer_metering_remaining_points")?
199                .clone();
200            fenv.as_mut(store).remaining_points = Some(g_1.clone());
201            let g_2 = instance
202                .exports
203                .get_global("wasmer_metering_points_exhausted")?
204                .clone();
205            fenv.as_mut(store).exhausted_points = Some(g_2.clone());
206
207            self.env.remaining_points = Some(g_1);
208            self.env.exhausted_points = Some(g_2);
209        }
210
211        Ok(())
212    }

352        interface: &dyn crate::Interface,
353        wat: &str,
354        gas_costs: GasCosts,
355    ) -> VMResult<(Store, ASContext, Instance)> {
356        let condom_limits = CondomLimits::default();
357        let limit = 10_000_000;

359        let module = Module::new(&engine, wat2wasm(wat.as_bytes()).unwrap()).unwrap();
